// ============ UTILITY FUNCTIONS ============
std::string difficultyToStr(Difficulty d) {
    switch (d) {
    case Difficulty::BEGINNER: return "Beginner";
    case Difficulty::INTERMEDIATE: return "Intermediate";
    case Difficulty::ADVANCED: return "Advanced";
    }
    return "";
}

Difficulty strToDifficulty(const std::string& s) {
    if (s == "Beginner") return Difficulty::BEGINNER;
    if (s == "Intermediate") return Difficulty::INTERMEDIATE;
    if (s == "Advanced") return Difficulty::ADVANCED;
    return Difficulty::BEGINNER;
}

std::vector<std::string> split(const std::string& str, char delimiter) {
    std::vector<std::string> tokens;
    std::stringstream ss(str);
    std::string token;
    while (std::getline(ss, token, delimiter)) {
        if (!token.empty()) tokens.push_back(token);
    }
    return tokens;
}

std::string trim(const std::string& str) {
    size_t first = str.find_first_not_of(" \t\n\r");
    if (first == std::string::npos) return "";
    size_t last = str.find_last_not_of(" \t\n\r");
    return str.substr(first, (last - first + 1));
}

// ============ COURSE CLASS ============
class Course {
public:
    int id;
    std::string name;
    Difficulty difficulty;
    int durationWeeks;
    std::string category;
    std::vector<int> prerequisites;
    std::vector<std::string> skills;
    double aiScore;
    bool completed;
    int hoursPerWeek;
    std::string instructor;
    std::string description;
    double rating;
    int enrollmentCount;

    Course() : id(0), difficulty(Difficulty::BEGINNER), durationWeeks(0),
        aiScore(0.0), completed(false), hoursPerWeek(5), rating(0.0), enrollmentCount(0) {
    }

    Course(int i, std::string n, Difficulty d, int dur, std::string cat, std::vector<int> prereqs,
        std::vector<std::string> sk, int hrs = 5, std::string inst = "Instructor")
        : id(i), name(n), difficulty(d), durationWeeks(dur), category(cat),
        prerequisites(prereqs), skills(sk), aiScore(0.0), completed(false),
        hoursPerWeek(hrs), instructor(inst), rating(0.0), enrollmentCount(0) {
        // Input validation
        if (hrs < 0 || hrs > 168) throw std::invalid_argument("Invalid hours per week");
        if (dur < 0) throw std::invalid_argument("Invalid duration");
    }

    void display() const {
        std::cout << "  [" << id << "] " << name << " (" << difficultyToStr(difficulty) << ")\n";
        std::cout << "      Category: " << category << " | Duration: " << durationWeeks << " weeks\n";
        std::cout << "      Hours/Week: " << hoursPerWeek << " | Instructor: " << instructor << "\n";
        std::cout << "      AI Score: " << std::fixed << std::setprecision(2) << aiScore;
        std::cout << " | Rating: " << rating << "/5.0 | Enrolled: " << enrollmentCount << "\n";
        if (!description.empty()) {
            std::cout << "      Description: " << description << "\n";
        }
    }

    std::string serialize() const {
        std::stringstream ss;
        ss << id << "|" << name << "|" << (int)difficulty << "|" << durationWeeks << "|"
            << category << "|";

        for (size_t i = 0; i < prerequisites.size(); i++) {
            ss << prerequisites[i];
            if (i < prerequisites.size() - 1) ss << ",";
        }
        ss << "|";

        for (size_t i = 0; i < skills.size(); i++) {
            ss << skills[i];
            if (i < skills.size() - 1) ss << ",";
        }
        ss << "|" << hoursPerWeek << "|" << instructor << "|" << aiScore << "|"
            << completed << "|" << description << "|" << rating << "|" << enrollmentCount;

        return ss.str();
    }

    static std::unique_ptr<Course> deserialize(const std::string& line) {
        try {
            std::vector<std::string> parts = split(line, '|');
            if (parts.size() < 14) {
                throw std::runtime_error("Invalid course data format: insufficient fields");
            }

            auto c = std::make_unique<Course>();
            c->id = std::stoi(parts[0]);
            c->name = parts[1];
            c->difficulty = (Difficulty)std::stoi(parts[2]);
            c->durationWeeks = std::stoi(parts[3]);
            c->category = parts[4];

            if (!parts[5].empty()) {
                std::vector<std::string> prereqs = split(parts[5], ',');
                for (const std::string& p : prereqs) {
                    c->prerequisites.push_back(std::stoi(p));
                }
            }

            if (!parts[6].empty()) {
                c->skills = split(parts[6], ',');
            }

            c->hoursPerWeek = std::stoi(parts[7]);
            c->instructor = parts[8];
            c->aiScore = std::stod(parts[9]);
            c->completed = std::stoi(parts[10]);
            c->description = parts[11];
            c->rating = std::stod(parts[12]);
            c->enrollmentCount = std::stoi(parts[13]);

            // Validation
            if (c->hoursPerWeek < 0 || c->hoursPerWeek > 168) {
                throw std::runtime_error("Invalid hours per week in course data");
            }

            return c;
        } catch (const std::exception& e) {
            std::cerr << "Error deserializing course: " << e.what() << std::endl;
            return nullptr;
        }
    }
};

// ============ SCHEDULE SLOT CLASS ============
class ScheduleSlot {
public:
    ScheduleDay day;
    int startHour;
    int duration;
    int courseId;
    std::string courseName;
    std::string activity;

    ScheduleSlot(ScheduleDay d, int start, int dur, int cid, std::string cname, std::string act)
        : day(d), startHour(start), duration(dur), courseId(cid),
        courseName(cname), activity(act) {
        // Validation
        if (start < 0 || start > 23) throw std::invalid_argument("Invalid start hour");
        if (dur < 0 || start + dur > 24) throw std::invalid_argument("Invalid duration");
    }

    void display() const {
        std::cout << "    " << dayToStr(day) << " "
            << std::setfill('0') << std::setw(2) << startHour << ":00 - "
            << std::setw(2) << (startHour + duration) << ":00"
            << " | " << courseName << " (" << activity << ")\n";
    }

    std::string serialize() const {
        std::stringstream ss;
        ss << (int)day << "," << startHour << "," << duration << ","
            << courseId << "," << courseName << "," << activity;
        return ss.str();
    }

    static ScheduleSlot deserialize(const std::string& line) {
        std::vector<std::string> parts = split(line, ',');
        if (parts.size() < 6) {
            throw std::runtime_error("Invalid schedule slot data");
        }
        return ScheduleSlot(
            (ScheduleDay)std::stoi(parts[0]),
            std::stoi(parts[1]),
            std::stoi(parts[2]),
            std::stoi(parts[3]),
            parts[4],
            parts[5]
        );
    }
};

// ============ WEEKLY SCHEDULE CLASS ============
class WeeklySchedule {
private:
    std::vector<ScheduleSlot> slots;
    std::map<ScheduleDay, int> dailyHours;
    int totalHoursPerWeek;

public:
    WeeklySchedule() : totalHoursPerWeek(0) {
        for (int i = 0; i < 7; i++) {
            dailyHours[(ScheduleDay)i] = 0;
        }
    }

    bool addSlot(const ScheduleSlot& slot) {
        // Fixed overlap detection
        for (const auto& existing : slots) {
            if (existing.day == slot.day) {
                int existingEnd = existing.startHour + existing.duration;
                int newEnd = slot.startHour + slot.duration;

                // Proper overlap check: overlaps if start < existingEnd AND end > existingStart
                if (slot.startHour < existingEnd && newEnd > existing.startHour) {
                    return false;  // Overlap detected
                }
            }
        }

        slots.push_back(slot);
        dailyHours[slot.day] += slot.duration;
        totalHoursPerWeek += slot.duration;
        return true;
    }

    void display() const {
        std::cout << "\n========== WEEKLY SCHEDULE ==========\n";
        std::cout << "Total Hours: " << totalHoursPerWeek << " hours/week\n\n";

        for (int i = 0; i < 7; i++) {
            ScheduleDay day = (ScheduleDay)i;
            std::cout << "  " << dayToStr(day) << " (" << dailyHours.at(day) << " hours):\n";

            bool hasSlots = false;
            for (const auto& slot : slots) {
                if (slot.day == day) {
                    slot.display();
                    hasSlots = true;
                }
            }
            if (!hasSlots) {
                std::cout << "    No classes scheduled\n";
            }
            std::cout << "\n";
        }
        std::cout << "====================================\n\n";
    }

    int getTotalHours() const { return totalHoursPerWeek; }
    std::vector<ScheduleSlot> getSlots() const { return slots; }
    
    void clear() {
        slots.clear();
        totalHoursPerWeek = 0;
        for (int i = 0; i < 7; i++) {
            dailyHours[(ScheduleDay)i] = 0;
        }
    }

    std::string serialize() const {
        std::stringstream ss;
        ss << slots.size() << "\n";
        for (const auto& slot : slots) {
            ss << slot.serialize() << "\n";
        }
        return ss.str();
    }

    void deserialize(std::ifstream& file) {
        clear();
        int count;
        file >> count;
        file.ignore();

        for (int i = 0; i < count; i++) {
            std::string line;
            std::getline(file, line);
            if (!line.empty()) {
                try {
                    ScheduleSlot slot = ScheduleSlot::deserialize(line);
                    addSlot(slot);
                } catch (const std::exception& e) {
                    std::cerr << "Error deserializing slot: " << e.what() << std::endl;
                }
            }
        }
    }
};

// ============ AVL TREE NODE ============
class AVLNode {
public:
    Course* course;
    AVLNode* left;
    AVLNode* right;
    int height;

    AVLNode(Course* c) : course(c), left(nullptr), right(nullptr), height(1) {}
};

// ============ AVL TREE CLASS ============
class AVLTree {
private:
    AVLNode* root;
    bool ownsData;  // Flag to track ownership

    int getHeight(AVLNode* node) const { return node ? node->height : 0; }
    int getBalance(AVLNode* node) const { return node ? getHeight(node->left) - getHeight(node->right) : 0; }
    void updateHeight(AVLNode* node) {
        if (node) node->height = 1 + std::max(getHeight(node->left), getHeight(node->right));
    }

    AVLNode* rotateRight(AVLNode* y) {
        AVLNode* x = y->left;
        AVLNode* T2 = x->right;
        x->right = y;
        y->left = T2;
        updateHeight(y);
        updateHeight(x);
        return x;
    }

    AVLNode* rotateLeft(AVLNode* x) {
        AVLNode* y = x->right;
        AVLNode* T2 = y->left;
        y->left = x;
        x->right = T2;
        updateHeight(x);
        updateHeight(y);
        return y;
    }

    AVLNode* insertRecursive(AVLNode* node, Course* course) {
        if (!node) return new AVLNode(course);

        if (course->aiScore < node->course->aiScore)
            node->left = insertRecursive(node->left, course);
        else if (course->aiScore > node->course->aiScore)
            node->right = insertRecursive(node->right, course);
        else
            return node;

        updateHeight(node);
        int balance = getBalance(node);

        if (balance > 1 && course->aiScore < node->left->course->aiScore)
            return rotateRight(node);
        if (balance < -1 && course->aiScore > node->right->course->aiScore)
            return rotateLeft(node);
        if (balance > 1 && course->aiScore > node->left->course->aiScore) {
            node->left = rotateLeft(node->left);
            return rotateRight(node);
        }
        if (balance < -1 && course->aiScore < node->right->course->aiScore) {
            node->right = rotateRight(node->right);
            return rotateLeft(node);
        }

        return node;
    }

    void inorderTraversal(AVLNode* node, std::vector<Course*>& result) const {
        if (!node) return;
        inorderTraversal(node->right, result);
        result.push_back(node->course);
        inorderTraversal(node->left, result);
    }

    void deleteTree(AVLNode* node, bool deleteCourses) {
        if (!node) return;
        deleteTree(node->left, deleteCourses);
        deleteTree(node->right, deleteCourses);
        if (deleteCourses && node->course) {
            delete node->course;
        }
        delete node;
    }

public:
    AVLTree(bool ownsData = false) : root(nullptr), ownsData(ownsData) {}
    ~AVLTree() { deleteTree(root, ownsData); }

    void insert(Course* course) { root = insertRecursive(root, course); }

    std::vector<Course*> getSortedCourses() const {
        std::vector<Course*> result;
        inorderTraversal(root, result);
        return result;
    }

    void clear() {
        deleteTree(root, ownsData);
        root = nullptr;
    }
};

// ============ BST NODE ============
class BSTNode {
public:
    int key;
    Course* course;
    BSTNode* left;
    BSTNode* right;

    BSTNode(int k, Course* c) : key(k), course(c), left(nullptr), right(nullptr) {}
};

// ============ BST CLASS ============
class BST {
private:
    BSTNode* root;
    bool ownsData;

    BSTNode* insertRecursive(BSTNode* node, int key, Course* course) {
        if (!node) return new BSTNode(key, course);
        if (key < node->key) node->left = insertRecursive(node->left, key, course);
        else if (key > node->key) node->right = insertRecursive(node->right, key, course);
        return node;
    }

    Course* searchRecursive(BSTNode* node, int key) const {
        if (!node) return nullptr;
        if (key == node->key) return node->course;
        if (key < node->key) return searchRecursive(node->left, key);
        return searchRecursive(node->right, key);
    }

    void deleteTree(BSTNode* node, bool deleteCourses) {
        if (!node) return;
        deleteTree(node->left, deleteCourses);
        deleteTree(node->right, deleteCourses);
        if (deleteCourses && node->course) {
            delete node->course;
        }
        delete node;
    }

public:
    BST(bool ownsData = false) : root(nullptr), ownsData(ownsData) {}
    ~BST() { deleteTree(root, ownsData); }

    void insert(int key, Course* course) { root = insertRecursive(root, key, course); }
    Course* search(int courseId) const { return searchRecursive(root, courseId); }

    void clear() {
        deleteTree(root, ownsData);
        root = nullptr;
    }
};

// ============ GRAPH CLASS ============
class Graph {
private:
    std::map<int, std::vector<int>> adjList;
    std::map<int, std::vector<int>> prerequisites;

    bool hasCycleDFS(int node, std::map<int, int>& visited) const {
        visited[node] = 1;
        for (int neighbor : adjList.at(node)) {
            if (visited[neighbor] == 1) return true;
            if (visited[neighbor] == 0 && hasCycleDFS(neighbor, visited)) return true;
        }
        visited[node] = 2;
        return false;
    }

public:
    void addCourse(int courseId) {
        if (adjList.find(courseId) == adjList.end()) {
            adjList[courseId] = std::vector<int>();
            prerequisites[courseId] = std::vector<int>();
        }
    }

    void addPrerequisite(int courseId, int prereqId) {
        prerequisites[courseId].push_back(prereqId);
        adjList[prereqId].push_back(courseId);
    }

    bool hasCycle() const {
        std::map<int, int> visited;
        for (const auto& pair : adjList) {
            if (visited[pair.first] == 0) {
                if (hasCycleDFS(pair.first, visited)) return true;
            }
        }
        return false;
    }

    // Fixed: Use unordered_set for O(1) lookup
    bool canTakeCourse(int courseId, const std::unordered_set<int>& completedSet) const {
        auto it = prerequisites.find(courseId);
        if (it == prerequisites.end()) return true;
        
        for (int prereq : it->second) {
            if (completedSet.find(prereq) == completedSet.end()) {
                return false;
            }
        }
        return true;
    }

    std::vector<int> getPrerequisites(int courseId) const { 
        auto it = prerequisites.find(courseId);
        return it != prerequisites.end() ? it->second : std::vector<int>();
    }
    
    std::vector<int> getNextCourses(int courseId) const { 
        auto it = adjList.find(courseId);
        return it != adjList.end() ? it->second : std::vector<int>();
    }

    void clear() {
        adjList.clear();
        prerequisites.clear();
    }
};

// ============ LINKED LIST NODE ============
class ListNode {
public:
    int courseId;
    std::string courseName;
    double score;
    time_t completionDate;
    ListNode* next;
    ListNode* prev;

    ListNode(int id, std::string name, double s)
        : courseId(id), courseName(name), score(s), next(nullptr), prev(nullptr) {
        time(&completionDate);
    }
};

// ============ LINKED LIST CLASS ============
class LinkedList {
private:
    ListNode* head;
    ListNode* tail;
    int size;

public:
    LinkedList() : head(nullptr), tail(nullptr), size(0) {}

    ~LinkedList() {
        ListNode* current = head;
        while (current) {
            ListNode* temp = current;
            current = current->next;
            delete temp;
        }
    }

    // Delete copy constructor and assignment operator
    LinkedList(const LinkedList&) = delete;
    LinkedList& operator=(const LinkedList&) = delete;

    void append(int courseId, std::string courseName, double score) {
        ListNode* newNode = new ListNode(courseId, courseName, score);
        if (!head) {
            head = tail = newNode;
        }
        else {
            tail->next = newNode;
            newNode->prev = tail;
            tail = newNode;
        }
        size++;
    }

    void display() const {
        std::cout << "\n========== Completed Courses History ==========\n";
        ListNode* current = head;
        int count = 1;
        while (current) {
            std::cout << count++ << ". " << current->courseName
                << " (ID: " << current->courseId << ") - Score: "
                << std::fixed << std::setprecision(1) << current->score << "/100\n";
            current = current->next;
        }
        if (size == 0) std::cout << "No completed courses yet.\n";
        std::cout << "=============================================\n\n";
    }

    double getAverageScore() const {
        if (size == 0) return 0.0;
        double total = 0.0;
        ListNode* current = head;
        while (current) {
            total += current->score;
            current = current->next;
        }
        return total / size;
    }

    int getSize() const { return size; }

    std::string serialize() const {
        std::stringstream ss;
        ss << size << "\n";
        ListNode* current = head;
        while (current) {
            ss << current->courseId << "," << current->courseName << ","
                << current->score << "," << current->completionDate << "\n";
            current = current->next;
        }
        return ss.str();
    }

    void deserialize(std::ifstream& file) {
        int count;
        file >> count;
        file.ignore();

        for (int i = 0; i < count; i++) {
            std::string line;
            std::getline(file, line);
            if (!line.empty()) {
                std::vector<std::string> parts = split(line, ',');
                if (parts.size() >= 3) {
                    append(std::stoi(parts[0]), parts[1], std::stod(parts[2]));
                }
            }
        }
    }

    void clear() {
        ListNode* current = head;
        while (current) {
            ListNode* temp = current;
            current = current->next;
            delete temp;
        }
        head = tail = nullptr;
        size = 0;
    }
};

// ============ PATH STACK CLASS ============
class PathStack {
private:
    std::stack<int> pathStack;
    std::vector<int> currentPath;

public:
    void push(int courseId) {
        pathStack.push(courseId);
        currentPath.push_back(courseId);
    }

    int pop() {
        if (!pathStack.empty()) {
            int courseId = pathStack.top();
            pathStack.pop();
            if (!currentPath.empty()) currentPath.pop_back();
            return courseId;
        }
        return -1;
    }

    bool isEmpty() const { return pathStack.empty(); }
    std::vector<int> getPath() const { return currentPath; }
    
    void clear() {
        while (!pathStack.empty()) pathStack.pop();
        currentPath.clear();
    }
};

// ============ DEPTH LIMITED SEARCH CLASS ============
class DepthLimitedSearch {
private:
    const Graph* graph;
    const std::map<int, Course*>* courseMap;

    void generatePathRecursive(int courseId, int depth, int limit, std::vector<int>& path,
        std::map<int, bool>& visited, std::unordered_set<int>& completedSet) const {
        if (depth > limit || path.size() >= 10) return;

        if (!visited[courseId] && graph->canTakeCourse(courseId, completedSet)) {
            visited[courseId] = true;
            path.push_back(courseId);
            completedSet.insert(courseId);  // Add to set for checking

            for (int nextId : graph->getNextCourses(courseId)) {
                if (!visited[nextId]) {
                    generatePathRecursive(nextId, depth + 1, limit, path, visited, completedSet);
                }
            }
        }
    }

public:
    DepthLimitedSearch(const Graph* g, const std::map<int, Course*>* cm) : graph(g), courseMap(cm) {}

    // Fixed: Does not modify input completedCourses
    std::vector<int> generateLimitedPath(int startId, int depthLimit, const std::vector<int>& completedCourses) const {
        std::vector<int> path;
        std::map<int, bool> visited;
        std::unordered_set<int> completedSet(completedCourses.begin(), completedCourses.end());

        for (int id : completedCourses) visited[id] = true;

        generatePathRecursive(startId, 0, depthLimit, path, visited, completedSet);
        return path;
    }
};

// ============ AI ENGINE CLASS ============
class AIEngine {
public:
    static double calculateAIScore(const Course* course, const std::map<std::string, int>& studentSkills,
        SkillLevel skillLevel, const std::string& learningGoal,
        int availableHours, const std::unordered_set<int>& completedSet) {
        double score = 0.0;

        // Goal alignment (35%)
        if (course->category == learningGoal) score += 35.0;
        else if (course->category.find(learningGoal) != std::string::npos) score += 20.0;

        // Difficulty match (30%)
        switch (skillLevel) {
        case SkillLevel::NOVICE:
            if (course->difficulty == Difficulty::BEGINNER) score += 30.0;
            else if (course->difficulty == Difficulty::INTERMEDIATE) score += 10.0;
            break;
        case SkillLevel::INTERMEDIATE_LEARNER:
            if (course->difficulty == Difficulty::BEGINNER) score += 15.0;
            else if (course->difficulty == Difficulty::INTERMEDIATE) score += 30.0;
            else if (course->difficulty == Difficulty::ADVANCED) score += 10.0;
            break;
        case SkillLevel::PROFICIENT:
            if (course->difficulty == Difficulty::INTERMEDIATE) score += 20.0;
            else if (course->difficulty == Difficulty::ADVANCED) score += 30.0;
            break;
        case SkillLevel::EXPERT:
            if (course->difficulty == Difficulty::ADVANCED) score += 30.0;
            else score += 15.0;
            break;
        }

        // Time feasibility (20%) - Fixed overflow issue
        if (course->hoursPerWeek <= availableHours) score += 20.0;
        else if (availableHours > 0 && course->hoursPerWeek <= static_cast<double>(availableHours) * 1.5) score += 10.0;

        // Prerequisites completion (15%)
        bool allPrereqsComplete = true;
        for (int prereq : course->prerequisites) {
            if (completedSet.find(prereq) == completedSet.end()) {
                allPrereqsComplete = false;
                break;
            }
        }
        if (allPrereqsComplete) score += 15.0;

        return score;
    }
};

// ============ STUDENT CLASS ============
class Student {
public:
    int id;
    std::string name;
    std::string email;
    SkillLevel skillLevel;
    std::string learningGoal;
    int availableHours;
    std::vector<int> completedCourses;
    LinkedList completedHistory;
    std::map<std::string, int> skills;
    WeeklySchedule schedule;
    int stressLevel;
    std::vector<std::string> concerns;
    std::string mentalHealthNote;

    Student() : id(0), skillLevel(SkillLevel::NOVICE), availableHours(10), stressLevel(5) {}

    Student(int i, std::string n, std::string e, SkillLevel level, std::string goal, int hours)
        : id(i), name(n), email(e), skillLevel(level), learningGoal(goal),
        availableHours(hours), stressLevel(5) {
        if (hours < 0 || hours > 168) throw std::invalid_argument("Invalid available hours");
    }

    void completeCourse(int courseId, const std::string& courseName, double score, const std::vector<std::string>& courseSkills) {
        completedCourses.push_back(courseId);
        completedHistory.append(courseId, courseName, score);

        for (const std::string& skill : courseSkills) {
            if (skills.find(skill) == skills.end()) {
                skills[skill] = 30;
            }
            else {
                skills[skill] = std::min(100, skills[skill] + 20);
            }
        }

        size_t count = completedCourses.size();
        if (count >= 3 && skillLevel == SkillLevel::NOVICE) {
            skillLevel = SkillLevel::INTERMEDIATE_LEARNER;
            std::cout << "✨ Skill level upgraded to: Intermediate!\n";
        }
        else if (count >= 6 && skillLevel == SkillLevel::INTERMEDIATE_LEARNER) {
            skillLevel = SkillLevel::PROFICIENT;
            std::cout << "✨ Skill level upgraded to: Proficient!\n";
        }
        else if (count >= 10 && skillLevel == SkillLevel::PROFICIENT) {
            skillLevel = SkillLevel::EXPERT;
            std::cout << "✨ Skill level upgraded to: Expert!\n";
        }
    }

    void displayProfile() const {
        std::cout << "\n========== STUDENT PROFILE ==========\n";
        std::cout << "ID: " << id << "\n";
        std::cout << "Name: " << name << "\n";
        std::cout << "Email: " << email << "\n";
        std::cout << "Skill Level: " << skillLevelToStr(skillLevel) << "\n";
        std::cout << "Learning Goal: " << learningGoal << "\n";
        std::cout << "Available Time: " << availableHours << " hours/week\n";
        std::cout << "Completed Courses: " << completedCourses.size() << "\n";
        std::cout << "Stress Level: " << stressLevel << "/10\n";

        if (!skills.empty()) {
            std::cout << "\nSkills:\n";
            for (const auto& pair : skills) {
                std::cout << "  - " << pair.first << ": " << pair.second << "%\n";
            }
        }
        std::cout << "====================================\n\n";
    }

    ProgressReport generateReport(int totalCourses) const {
        ProgressReport report;
        report.studentName = name;
        report.completedCourses = completedCourses.size();
        report.totalCourses = totalCourses;
        report.averageScore = completedHistory.getAverageScore();
        report.totalStudyHours = schedule.getTotalHours() * 4;
        report.stressLevel = stressLevel;
        report.mentalHealthNote = mentalHealthNote;

        for (const auto& pair : skills) {
            report.completedSkills.push_back(pair.first + " (" + std::to_string(pair.second) + "%)");
        }

        if (stressLevel >= 7) {
            report.recommendations.push_back("⚠️ Consider reducing course load");
            report.recommendations.push_back("🧘 Schedule breaks between study sessions");
        }
        if (completedHistory.getAverageScore() < 70) {
            report.recommendations.push_back("📚 Review fundamental concepts");
            report.recommendations.push_back("👥 Seek tutoring or study groups");
        }
        if (schedule.getTotalHours() > availableHours) {
            report.recommendations.push_back("⏰ Adjust schedule to match available time");
        }

        return report;
    }

    void saveToFile(const std::string& filename) const {
        std::ofstream file(filename);
        if (!file.is_open()) {
            throw std::runtime_error("Cannot open file: " + filename);
        }

        file << id << "\n" << name << "\n" << email << "\n"
            << skillLevelToStr(skillLevel) << "\n" << learningGoal << "\n"
            << availableHours << "\n" << stressLevel << "\n";

        file << completedCourses.size() << "\n";
        for (int cid : completedCourses) {
            file << cid << " ";
        }
        file << "\n";

        file << skills.size() << "\n";
        for (const auto& pair : skills) {
            file << pair.first << "," << pair.second << "\n";
        }

        file << mentalHealthNote << "\n";
        file << completedHistory.serialize();
        file << schedule.serialize();

        file.close();
    }

    static std::unique_ptr<Student> loadFromFile(const std::string& filename) {
        std::ifstream file(filename);
        if (!file.is_open()) {
            throw std::runtime_error("Cannot open file: " + filename);
        }

        auto s = std::make_unique<Student>();
        std::string line;

        file >> s->id;
        file.ignore();
        std::getline(file, s->name);
        std::getline(file, s->email);
        std::getline(file, line);
        s->skillLevel = strToSkillLevel(line);
        std::getline(file, s->learningGoal);
        file >> s->availableHours >> s->stressLevel;
        file.ignore();

        int completedCount;
        file >> completedCount;
        s->completedCourses.resize(completedCount);
        for (int i = 0; i < completedCount; i++) {
            file >> s->completedCourses[i];
        }
        file.ignore();

        int skillCount;
        file >> skillCount;
        file.ignore();
        for (int i = 0; i < skillCount; i++) {
            std::getline(file, line);
            std::vector<std::string> parts = split(line, ',');
            if (parts.size() == 2) {
                s->skills[parts[0]] = std::stoi(parts[1]);
            }
        }

        std::getline(file, s->mentalHealthNote);
        s->completedHistory.deserialize(file);
        s->schedule.deserialize(file);

        file.close();
        return s;
    }
};

// ============ SCHEDULE GENERATOR CLASS ============
class ScheduleGenerator {
private:
    WeeklySchedule* schedule;
    std::vector<Course*> courses;

public:
    ScheduleGenerator(WeeklySchedule* sched) : schedule(sched) {}

    void addCourse(Course* course) {
        courses.push_back(course);
    }

    bool generateSchedule(int maxHoursPerWeek) {
        schedule->clear();
        int totalHours = 0;

        // Simple greedy scheduling algorithm
        for (Course* course : courses) {
            if (totalHours + course->hoursPerWeek > maxHoursPerWeek) {
                continue;  // Skip if exceeds max hours
            }

            // Try to schedule on weekdays
            for (int day = 0; day < 5 && totalHours < maxHoursPerWeek; day++) {
                int hoursToSchedule = std::min(course->hoursPerWeek, 3);  // Max 3 hours per day
                
                try {
                    ScheduleSlot slot(
                        (ScheduleDay)day,
                        9 + (day * 2) % 8,  // Vary start times
                        hoursToSchedule,
                        course->id,
                        course->name,
                        "Lecture"
                    );

                    if (schedule->addSlot(slot)) {
                        totalHours += hoursToSchedule;
                    }
                } catch (const std::exception& e) {
                    std::cerr << "Error scheduling: " << e.what() << std::endl;
                }
            }
        }

        return totalHours > 0;
    }

    void clear() {
        courses.clear();
    }
};
